# Dependency overrides for the Modal image build, passed explicitly as
# `uv pip install --override modal-overrides.txt`.
#
# pyproject.toml's [tool.uv] override-dependencies carries the same flax pin,
# but the image build demonstrably does not honor it: builds on 2026-08-26
# fetched https://github.com/google/flax at HEAD (01854da, 0.12.9, which
# requires Python>=3.12) and failed on the image's Python 3.11, despite the
# pin being present and unchanged. `uv pip install` is the pip-compatible
# interface and does not read the project's [tool.uv] table the way
# `uv sync`/`uv lock` do -- so the pin has to be handed to it directly.
#
# t5x declares its own UNPINNED flax URL, which is what pulls HEAD in; an
# override is the only thing that outranks a transitive URL requirement.
#
# Keep this in sync with pyproject.toml's [tool.uv.sources] flax rev.
flax @ git+https://github.com/google/flax@c31deb2b8ec8d0cc1b2092aa81890823d21a4739

# Same problem, same fix, for seqio (hit 2026-09-12): t5x also declares an
# unpinned seqio URL, which conflicts with pyproject.toml's pinned rev during
# the Modal image build ("Requirements contain conflicting URLs for package
# `seqio`"). Keep in sync with pyproject.toml's [tool.uv.sources] seqio rev.
seqio @ git+https://github.com/google/seqio@aaa0e111c5e9d8fb49bfa992f2272a121a95f741

# `t5` leaves `transformers` effectively unbounded below, so the resolver is
# free to backtrack into 2021-era releases. A rebuild on 2026-10-04 did
# exactly that -- transformers 4.12.2, which pins tokenizers 0.10.3, which
# publishes no cp311 wheel -- so the build tried to compile tokenizers from
# source and died for want of a Rust toolchain. Nothing about the image
# changed to cause it: `add_local_dir(copy=True)` sits above `uv pip install`,
# so editing any file in the fork invalidates that layer and re-resolves the
# whole graph from scratch, which is when the drift surfaces.
#
# Pinned to the versions uv.lock already resolves locally (a working install),
# rather than to a floor, because resolver drift between otherwise identical
# builds is the recurring failure mode here. Keep in sync with uv.lock.
transformers==4.57.6
tokenizers==0.22.2
